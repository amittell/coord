# Pinned dependency for the OPTIONAL PostgreSQL backend
# (coordination/pg_backend.py, selected via COORD_DATABASE_URL).
#
# Kept out of the base requirements.txt so a SQLite-only install/build
# carries no Postgres driver. The container image installs this layer on
# top of requirements.txt (see Dockerfile) so the SAME release image can
# be pointed at Postgres at cutover time just by setting
# COORD_DATABASE_URL (deploy/k8s/ha-cutover/deployment.yaml depends on
# this; an image built without it CrashLoops at startup the moment
# COORD_DATABASE_URL is set). PyPI users get the unpinned equivalent via
# the project's [postgres] extra (pip install coord-mcp-server[postgres]).
#
# asyncpg carries no runtime dependencies on the Pythons this image
# ships, so this file is a single pin. Regenerate by diffing a clean
# `pip install .[postgres]` freeze against requirements.txt:
#   python3.12 -m venv /tmp/coord-pg
#   /tmp/coord-pg/bin/pip install '.[postgres]'
#   /tmp/coord-pg/bin/pip freeze | grep -vxF -f requirements.txt
asyncpg==0.31.0
